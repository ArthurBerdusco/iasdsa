-- -- Create cultos table
CREATE TABLE IF NOT EXISTS cultos (
  id SERIAL PRIMARY KEY,
  titulo VARCHAR(100) NOT NULL,
  diaSemana VARCHAR(20) NOT NULL,
  data VARCHAR(10) NOT NULL,
  hora VARCHAR(10) NOT NULL,
  orador VARCHAR(100) NOT NULL,
  imagem VARCHAR(255) NOT NULL DEFAULT '/images/default-banner.jpg',
  oradorImagem VARCHAR(255) NOT NULL DEFAULT '/images/pastores/sem-imagem.jpg',
  corDestaque VARCHAR(20) NOT NULL DEFAULT 'primary',
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Create index for more efficient date queries
CREATE INDEX IF NOT EXISTS idx_cultos_data ON cultos(data);
